    Prisma Database Schema (schema.prisma) 

    model User {
  id        String     @id @default(uuid())
  email     String     @unique
  name      String
  alterEgo  AlterEgo?
  createdAt DateTime   @default(now())
}

model AlterEgo {
  id          String       @id @default(uuid())
  userId      String       @unique
  user        User         @relation(fields: [userId], references: [id])
  name        String
  description String
  traits      String[]
  rules       String[]
  habits      Habit[]
  goals       Goal[]
  reflections Reflection[]
}

model Habit {
  id          String   @id @default(uuid())
  alterEgoId  String
  title       String
  completed   Boolean  @default(false)
}

model Reflection {
  id          String   @id @default(uuid())
  alterEgoId  String
  intention   String
  action      String
  obstacle    String
  createdAt   DateTime @default(now())
}
